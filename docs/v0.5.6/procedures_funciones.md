CREATE DEFINER=`root`@`localhost` PROCEDURE `SOCYALPRO`.`AVANZAR_CONTADOR`(
  IN P_IDCONTADOR INTEGER,
  OUT P_CONTADOR INTEGER
)
BEGIN
  SELECT INTCONTADOR INTO P_CONTADOR
  FROM CORE_CONTADOR
  WHERE IDCONTADOR = P_IDCONTADOR
  FOR UPDATE;

  UPDATE CORE_CONTADOR
  SET INTCONTADOR = INTCONTADOR + 1
  WHERE IDCONTADOR = P_IDCONTADOR;
END
